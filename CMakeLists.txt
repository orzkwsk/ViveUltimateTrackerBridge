cmake_minimum_required(VERSION 3.20)
project(ViveUltimateTrackerBridge VERSION 0.1.0 LANGUAGES CXX)

# Configure from the exact source tree being built. Reconfigure after commits;
# dirty builds remain explicitly marked rather than claiming an exact mapping.
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE VUT_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" symbolic-ref --short -q HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE VUT_GIT_REF OUTPUT_STRIP_TRAILING_WHITESPACE)
if(VUT_GIT_REF STREQUAL "")
    set(VUT_GIT_REF "detached")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain --untracked-files=normal WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE VUT_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(VUT_GIT_DIRTY false)
if(NOT VUT_GIT_STATUS STREQUAL "")
    set(VUT_GIT_DIRTY true)
endif()
string(TIMESTAMP VUT_BUILD_TIMESTAMP "%Y-%m-%dT%H:%M:%SZ" UTC)
configure_file(cmake/diagnostic_build.hpp.in generated/vut/diagnostic_build.hpp @ONLY)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_protocol_v2.py" RESULT_VARIABLE revision_kit_result)
if(NOT revision_kit_result EQUAL 0)
 message(FATAL_ERROR "Protocol v2 fixed artifact verification failed")
endif()


set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(vut_backend STATIC
    src/protocol.cpp
    src/backend.cpp
)
target_include_directories(vut_backend PUBLIC include)
if(WIN32)
    target_compile_definitions(vut_backend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(vut_backend PRIVATE hid setupapi)
endif()

add_executable(vut_probe tools/vut_probe.cpp)
target_link_libraries(vut_probe PRIVATE vut_backend)

add_executable(vut_pair tools/vut_pair.cpp)
target_link_libraries(vut_pair PRIVATE vut_backend)
