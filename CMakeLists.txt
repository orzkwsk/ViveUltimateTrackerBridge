cmake_minimum_required(VERSION 3.20)
project(ViveUltimateTrackerBridge VERSION 0.1.0 LANGUAGES CXX)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_protocol_v2.py" RESULT_VARIABLE revision_kit_result)
if(NOT revision_kit_result EQUAL 0)
 message(FATAL_ERROR "Protocol v2 fixed artifact verification failed")
endif()


set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(vut_backend STATIC
    src/protocol.cpp
    src/backend.cpp
)
target_include_directories(vut_backend PUBLIC include)
if(WIN32)
    target_compile_definitions(vut_backend PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(vut_backend PRIVATE hid setupapi)
endif()

add_executable(vut_probe tools/vut_probe.cpp)
target_link_libraries(vut_probe PRIVATE vut_backend)

include(CTest)
if(BUILD_TESTING)
 add_test(NAME protocol_v2_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_protocol_v2.py")
endif()
if(BUILD_TESTING)
    add_executable(vut_protocol_tests tests/protocol_tests.cpp)
    target_link_libraries(vut_protocol_tests PRIVATE vut_backend)
    add_test(NAME protocol COMMAND vut_protocol_tests)
    add_executable(vut_runtime_policy_tests tests/runtime_policy_tests.cpp)
    target_link_libraries(vut_runtime_policy_tests PRIVATE vut_backend)
    add_test(NAME runtime_policy COMMAND vut_runtime_policy_tests)
endif()

option(VUT_BUILD_OBSERVATION "Build fixed C1 adapter and publisher" ON)
if(VUT_BUILD_OBSERVATION)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/import_protocol_kit.py" --verify
        RESULT_VARIABLE kit_result)
    if(NOT kit_result EQUAL 0)
        message(FATAL_ERROR "Fixed Task 1 kit verification failed; import the exact handoff or use -DVUT_BUILD_OBSERVATION=OFF for diagnostics only")
    endif()
    add_subdirectory(third_party/monaka-protocol-v2/cpp EXCLUDE_FROM_ALL)
    add_library(vut_observation STATIC src/observation_adapter.cpp src/observation_publisher.cpp)
    target_include_directories(vut_observation PUBLIC include)
    target_link_libraries(vut_observation PUBLIC vut_backend MonakaProtocol::Codec)
    if(WIN32)
        target_compile_definitions(vut_observation PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(vut_observation PRIVATE ws2_32)
        add_executable(vut_service tools/vut_service.cpp)
        target_compile_definitions(vut_service PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(vut_service PRIVATE vut_observation ole32)
    endif()
endif()
if(VUT_BUILD_OBSERVATION AND BUILD_TESTING)
    add_library(c1_mock_consumer STATIC tests/mock_consumer.cpp)
    target_link_libraries(c1_mock_consumer PUBLIC MonakaProtocol::Codec)
    add_executable(vut_observation_tests tests/observation_tests.cpp)
    target_link_libraries(vut_observation_tests PRIVATE vut_observation c1_mock_consumer)
    target_compile_definitions(vut_observation_tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME observation COMMAND vut_observation_tests)
    add_test(NAME protocol_kit_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/import_protocol_kit.py" --verify)
endif()
if(VUT_BUILD_OBSERVATION AND BUILD_TESTING)
    add_test(NAME loopback_udp COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/udp_smoke.py" "$<TARGET_FILE:vut_observation_tests>")
    set_tests_properties(loopback_udp PROPERTIES TIMEOUT 60)
endif()
if(VUT_BUILD_OBSERVATION AND BUILD_TESTING)
    add_executable(vut_common_codec_runner third_party/monaka-protocol-v2/cpp/tests/runner_v2.cpp)
    target_link_libraries(vut_common_codec_runner PRIVATE MonakaProtocol::Codec)
    find_package(Java 17 COMPONENTS Runtime Development)
    if(Java_FOUND)
        add_test(NAME cross_language COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cross_language.py"
            "$<TARGET_FILE:vut_common_codec_runner>" "$<TARGET_FILE:vut_observation_tests>" "${Java_JAVA_EXECUTABLE}" "${Java_JAVAC_EXECUTABLE}")
        set_tests_properties(cross_language PROPERTIES TIMEOUT 180)
    endif()
endif()
if(VUT_BUILD_OBSERVATION AND BUILD_TESTING AND WIN32)
    add_test(NAME service_configuration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_config.py" "$<TARGET_FILE:vut_service>")
endif()
